########################################    Creación VLAN's #################################################

enable
configure terminal

hostname SwTulcan

vlan 24
name ADM_04

end

#######################################    Creación Pool DHCPv6 ##########################################

enable
configure terminal

ip routing
ipv6 unicast-routing
ipv6 dhcp pool ADDRESS_IPv6
dns-server 2800:A:B:50::10
domain-name www.unite.edu.ec

end

####################################    Asignar Gateways VLAN's ##############################################

enable
configure terminal

interface vlan 1
ip address 192.168.1.1 255.255.255.0
ipv6 address 2800:A:B:1::1/64
no shutdown

interface vlan 24
ip address 192.168.24.1 255.255.255.0
ipv6 address 2800:A:B:24::1/64
ip helper-address 192.168.50.13
ipv6 dhcp SERVER ADDRESS_IPv6
ipv6 nd other-config-flag
no shutdown

end

#########################################   Membresia VLAN's #######################################################

enable
configure terminal

interface range G1/0/1
switchport mode access
switchport access vlan 24

end

enable
configure terminal

interface range G1/0/2
switchport mode access
sw voice vlan 1

end

######################################################################################################

enable
configure terminal

spanning-tree mode rapid-pvst

interface range G1/0/1 - 2
spanning-tree portfast

interface range G1/0/23 - 24
spanning-tree portfast

end

####################################    Enrutamiento Con Router ####################################################

enable
configure terminal

interface G1/0/24
no switchport
ip address 200.2.4.2 255.255.255.0
no shutdown

ip route 0.0.0.0 0.0.0.0 200.2.4.1

end

enable
configure terminal

router eigrp 210

network 4.4.4.0         0.0.0.3
network 6.6.6.0         0.0.0.3
network 200.2.4.0       0.0.0.255
network 192.168.0.0     0.0.255.255

end
