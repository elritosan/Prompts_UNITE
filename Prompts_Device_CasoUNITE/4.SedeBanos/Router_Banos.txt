enable
configure terminal

ip dhcp pool VOIP
network 192.168.5.0 255.255.255.0
default-router 192.168.5.1
option 150 ip 192.168.5.1

interface F0/0
ip address 192.168.5.1 255.255.255.0
no shutdown

telephony-service
max-ephones 10
max-dn 10
ip source-address 192.168.5.1 port 2000
auto assign 1 to 10

end

enable
configure terminal

ephone-dn 1
number 0401
ephone-dn 2
number 0402

end

enable
configure terminal

dial-peer voice 100 voip
destination-pattern 00..
session target ipv4:192.168.1.1

end

######################################## Asignacion Direccion Routers #################################################

enable
configure terminal

hostname RBanos

interface S0/1/1
ip add 4.4.4.2 255.255.255.252
ipv6 add 2800:4:4:4::2/64
no shutdown 

interface S0/3/0
ip add 6.6.6.2 255.255.255.252
ipv6 add 2800:6:6:6::2/64
no shutdown

end

######################################## Asignacion Direccion Host #################################################

enable
configure terminal

interface F1/0
ip add 200.2.4.1 255.255.255.0
ipv6 add 2800:A:C:4::1/64
no shutdown

end

######################################## Configurar Protocolo EIGRP #################################################

enable
configure terminal

ip routing
ipv6 unicast-routing

router eigrp 210
eigrp router-id 4.4.4.4
network 4.4.4.0         0.0.0.3
network 6.6.6.0         0.0.0.3
network 200.2.4.0       0.0.0.255
network 192.168.0.0     0.0.255.255

ipv6 router eigrp 220
eigrp router-id 4.4.4.4
no shutdown

interface F1/0
ipv6 eigrp 220
no shutdown

interface S0/1/1
ipv6 eigrp 220
no shutdown

interface S0/3/0
ipv6 eigrp 220
no shutdown

end
