enable
configure terminal
ip dhcp pool VOIP
network 192.168.1.0 255.255.255.0
default-router 192.168.1.1
option 150 ip 192.168.1.1 
interface F0/0 
ip address 192.168.1.1 255.255.255.0 
no shutdown
end

enable
configure terminal
telephony-service
max-ephones 2
max-dn 2
ip source-address 192.168.1.1 port 2000
auto assign 1 to 2
end

enable
configure terminal
ephone-dn 1
number 0301
end

enable
configure terminal
ephone-dn 2
number 0302
end


######################################## Asignacion Direccion Routers #################################################

enable
configure terminal

hostname RCayambe

interface S0/1/0
ip add 3.3.3.2 255.255.255.252
ipv6 add 2800:3:3:3::2/64
no shutdown 

interface S0/3/0
ip add 6.6.6.1 255.255.255.252
ipv6 add 2800:6:6:6::1/64
no shutdown

end

######################################## Asignacion Direccion Host #################################################

enable
configure terminal

interface F0/0
ip add 200.2.3.1 255.255.255.0
ipv6 add 2800:A:C:3::1/64
no shutdown

end

######################################## Configurar Protocolo EIGRP #################################################

enable
configure terminal

ip routing
ipv6 unicast-routing

router eigrp 210
eigrp router-id 3.3.3.3
network 3.3.3.0         0.0.0.3
network 6.6.6.0         0.0.0.3
network 200.2.3.0       0.0.0.255
passive-interface F0/0

ipv6 router eigrp 220
eigrp router-id 3.3.3.3
no shutdown

interface F0/0
ipv6 eigrp 220
no shutdown

interface S0/1/0
ipv6 eigrp 220
no shutdown

interface S0/3/0
ipv6 eigrp 220
no shutdown

end
