#########################################   Para VOIP    ##################################################

enable
configure terminal

ip dhcp pool VOIP
network 192.168.1.0 255.255.255.0
default-router 192.168.1.1
option 150 ip 192.168.1.1

interface F0/0 
ip address 192.168.1.1 255.255.255.0 
no shutdown
end

telephony-service
max-ephones 20
max-dn 20
ip source-address 192.168.1.1 port 2000
auto assign 1 to 20

end

enable
configure terminal

ephone-dn 1
number 0001
ephone-dn 2
number 0002

end

enable
configure terminal

dial-peer voice 101 voip
destination-pattern 01..
session target ipv4:192.168.2.1

dial-peer voice 102 voip
destination-pattern 02..
session target ipv4:192.168.3.1

end

#############################################   Para SNMP [Router o Switch] (Read/Write)   ##############################################

configure terminal

snmp-server community prueba-ro ro
snmp-server community prueba-rw rw
logging on
logging 192.168.50.13
logging trap debugging

end

######################################## Configurar Protocolo OSPF #################################################

enable
configure terminal

ip routing
ipv6 unicast-routing

route ospf 110
router-id 8.8.8.8
network 1.1.1.0         0.0.0.3     area 0
network 2.2.2.0         0.0.0.3     area 0

network 200.2.8.0       0.0.0.255   area 0
network 192.168.0.0     0.0.255.255   area 0

ipv6 router ospf 120
router-id 8.8.8.8

interface F1/0
ipv6 ospf 120 area 0
no shutdown

interface S0/0/0
ipv6 ospf 120 area 0
no shutdown

interface S0/0/1
ipv6 ospf 120 area 0
no shutdown

end
