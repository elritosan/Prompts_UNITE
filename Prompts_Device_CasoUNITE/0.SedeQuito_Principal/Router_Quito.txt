#########################################   Para VOIP    ##################################################

enable
configure terminal
ip dhcp pool VOIP
network 192.168.1.0 255.255.255.0
default-router 192.168.1.1
option 150 ip 192.168.1.1

interface F0/0 
ip address 192.168.1.1 255.255.255.0 
no shutdown
end

enable
configure terminal
telephony-service
max-ephones 2
max-dn 2
ip source-address 192.168.1.1 port 2000
auto assign 1 to 2
end

enable
configure terminal
ephone-dn 1
number 0001
end

enable
configure terminal
ephone-dn 2
number 0002
end

no shutdown 

interface S0/0/1

no shutdown

no shutdown

interface S0/1/1

no shutdown

end

######################################## Asignacion Direccion Host #################################################

enable
configure terminal

interface F0/0 
ip address 192.168.1.1 255.255.255.0 
ipv6 add 2800:A:B:1::1/64
no shutdown

interface F0/1
ip address 192.168.50.5 255.255.255.0
ipv6 add 2800:A:B:50::1/64
no shutdown

end

######################################## Configurar Protocolo OSPF #################################################

enable
configure terminal

ip routing
ipv6 unicast-routing

route ospf 110
router-id 0.0.0.0
network 1.1.1.0         0.0.0.3     area 0
network 2.2.2.0         0.0.0.3     area 0
network 192.168.1.0     0.0.0.255   area 0
network 192.168.50.0    0.0.0.255   area 0
passive-interface F0/0
passive-interface F0/1

ipv6 router ospf 120
router-id 0.0.0.0

interface F0/0
ipv6 ospf 120 area 0
no shutdown

interface F0/1
ipv6 ospf 120 area 0
no shutdown

interface S0/0/0
ipv6 ospf 120 area 0
no shutdown

interface S0/0/1
ipv6 ospf 120 area 0
no shutdown

end

######################################## Configurar Protocolo EIGRP #################################################

enable
configure terminal

ip routing
ipv6 unicast-routing

router eigrp 210
eigrp router-id 0.0.0.0
network 3.3.3.0         0.0.0.3
network 4.4.4.0         0.0.0.3

ipv6 router eigrp 220
eigrp router-id 0.0.0.0
no shutdown

interface S0/1/0
ipv6 eigrp 220
no shutdown

interface S0/1/1
ipv6 eigrp 220
no shutdown

end
