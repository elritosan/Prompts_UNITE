#########################################   Para VOIP    ##################################################

enable
configure terminal
ip dhcp pool VOIP
network 192.168.1.0 255.255.255.0
default-router 192.168.1.1
option 150 ip 192.168.1.1

interface F0/0 
ip address 192.168.1.1 255.255.255.0 
no shutdown
end

enable
configure terminal
telephony-service
max-ephones 2
max-dn 2
ip source-address 192.168.1.1 port 2000
auto assign 1 to 2
end

enable
configure terminal
ephone-dn 1
number 0001
end

enable
configure terminal
ephone-dn 2
number 0002
end

#############################################   Para  NTP   ##############################################

enable
configure terminal

ntp server 192.168.50.10
end

#############################################   Para SNMP (Interfaz)   ##############################################

enable
configure terminal

interface F0/1
ip address 192.168.50.110 255.255.255.0
no shutdown

end

Nota1: IP del Computador para MIB es "192.168.50.20" (Port: 161, Version: v1)
Nota2: IP del Servidor "192.168.50.13"
Nota3: IP de la Interfaz Router "192.168.50.110"

#############################################   Para SNMP [Router o Switch] (Read/Write)   ##############################################

enable
configure terminal

snmp-server community prueba-ro ro
snmp-server community prueba-rw rw
logging on
logging 192.168.50.13
logging trap debugging

end
